# 골든타임 지킴이 - 손에 쥐는 마이크로비트 (2대: 왼손용 / 오른손용)
# python.microbit.org 에 붙여넣고 올리기
# 오른손용으로 올릴 때는 아래 HAND = "L" 을 HAND = "R" 로 바꾸기
#
# 보내는 메시지
#   D,손,처짐각,회전각   팔 처짐 검사 중 0.1초마다
#   T,손,누른횟수        손가락 두드리기 검사 중 버튼(왼손 B, 오른손 A)을 누를 때마다
from microbit import *
import radio
import math

HAND = "L"
RAD2DEG = 57.2958
# 두드리기 버튼: 왼손은 B(화면 오른쪽 버튼, 왼손 엄지가 닿기 쉬움), 오른손은 A(화면 왼쪽 버튼)
TAP_BTN = button_b if HAND == "L" else button_a

radio.config(group=23, length=64, power=7)   # 받는 쪽과 그룹 번호가 같아야 함
radio.on()
display.show(HAND)

mode = ""
last_send = 0
count = 0


def angles():
    # 중력 방향으로 기울기 계산: 화면을 위로 해서 평평하게 쥐면 둘 다 0도
    x = accelerometer.get_x()
    y = accelerometer.get_y()
    z = accelerometer.get_z()
    pitch = math.atan2(y, math.sqrt(x * x + z * z)) * RAD2DEG   # 앞뒤로 기울어짐 (팔 처짐)
    roll = math.atan2(x, math.sqrt(y * y + z * z)) * RAD2DEG    # 좌우로 돌아감 (손바닥 회전)
    return pitch, roll


while True:
    msg = radio.receive()
    if msg == "DRIFT":
        mode = "DRIFT"
        display.show(Image.SQUARE)
    elif msg == "TAP_" + HAND:
        mode = "TAP"
        count = 0
        TAP_BTN.get_presses()           # 이전에 눌린 횟수 비우기
        display.show(Image.TARGET)
    elif msg == "STOP":
        mode = ""
        display.show(HAND)

    if mode == "DRIFT" and running_time() - last_send >= 100:
        last_send = running_time()
        p, r = angles()
        radio.send("D," + HAND + "," + str(round(p, 1)) + "," + str(round(r, 1)))
    elif mode == "TAP":
        n = TAP_BTN.get_presses()
        if n:
            count += n
            radio.send("T," + HAND + "," + str(count))
            display.show(str(count % 10))
    sleep(10)

# 골든타임 지킴이 - 노트북에 USB로 꽂는 수신용 마이크로비트 (1대)
# 손 마이크로비트의 라디오 메시지 → 노트북(시리얼)으로 전달
# 노트북 명령(DRIFT, TAP_L, TAP_R, STOP) → 라디오로 손 마이크로비트에 전달
# 노트북 결과(OK, WARN, ALARM) → 화면 아이콘과 소리
from microbit import *
import radio
import music

radio.config(group=23, length=64, power=7)
radio.on()
display.show(Image.HEART)
buf = ""

while True:
    msg = radio.receive()
    if msg:
        print(msg)

    if uart.any():
        data = uart.read()
        if data:
            buf += str(data, "utf-8")
            while "\n" in buf:
                line, buf = buf.split("\n", 1)
                line = line.strip()
                if line == "ALARM":
                    display.show(Image.NO)
                    music.play(["C6:2", "R:1"] * 6, wait=False)
                elif line == "OK":
                    display.show(Image.YES)
                elif line == "WARN":
                    display.show(Image.CONFUSED)
                elif line:
                    radio.send(line)
    sleep(5)

  return { drop: Math.max(m.dropL, m.dropR), diff: Math.abs(m.dropL - m.dropR), rot: Math.max(m.rotL, m.rotR),
           ratio: fast ? Math.min(m.tapL, m.tapR) / fast : 0 };
}
function judge(m, hist) {
  const d = derive(m), base = hist.length >= CFG.BASE_MIN, flags = [];
  for (const [k, lim] of [['drop', CFG.DROP], ['diff', CFG.DIFF], ['rot', CFG.ROT]]) {
    let bad;
    if (base) { const v = hist.map(h => h[k]), mu = mean(v), sd = Math.max(pstdev(v), 2); bad = (d[k] - mu) / sd > CFG.Z || d[k] > lim * 2; }
    else bad = d[k] > lim;
    if (bad) flags.push(k);
  }
  let bad;
  if (base) { const v = hist.map(h => h.ratio), mu = mean(v), sd = Math.max(pstdev(v), 0.05); bad = (mu - d.ratio) / sd > CFG.Z || d.ratio < 0.5; }
  else bad = d.ratio < CFG.TAP;
  if (bad) flags.push('ratio');
  const weak = m.dropL >= m.dropR ? '왼' : '오른', rot = m.rotL >= m.rotR ? '왼' : '오른', slow = m.tapL <= m.tapR ? '왼' : '오른';
  const text = { drop: `${weak}팔 처짐 ${d.drop.toFixed(0)}°`, diff: `좌우 팔 높이 차이 ${d.diff.toFixed(0)}°`,
                 rot: `${rot}손 안쪽 회전 ${d.rot.toFixed(0)}°`, ratio: `${slow}손 두드리기 느림 (${Math.round(d.ratio * 100)}%)` };
  const reasons = flags.map(f => text[f]);
  if (!m.speechOk) reasons.unshift('말이 어눌함');
  if (!m.faceOk) reasons.unshift('얼굴 한쪽 처짐');
  const level = (!m.faceOk || !m.speechOk || flags.length >= 2) ? '위험' : flags.length ? '재검' : '정상';
  return { level, reasons, flags, d, base, mode: base ? `개인 기준선 · 정상 기록 ${hist.length}회` : '기본 기준 · 정상 기록 3회 미만' };
}
const toJamo = s => s.replace(/[^\p{L}\p{N}]/gu, '').normalize('NFD');
function matchLen(a, b) {
  if (!a.length || !b.length) return 0;
  let best = 0, ai = 0, bi = 0; const dp = new Array(b.length + 1).fill(0);
  for (let i = 1; i <= a.length; i++) {
    let prev = 0;
    for (let j = 1; j <= b.length; j++) {
      const tmp = dp[j]; dp[j] = a[i - 1] === b[j - 1] ? prev + 1 : 0;
      if (dp[j] > best) { best = dp[j]; ai = i - best; bi = j - best; }
      prev = tmp;
    }
  }
  if (!best) return 0;
  return best + matchLen(a.slice(0, ai), b.slice(0, bi)) + matchLen(a.slice(ai + best), b.slice(bi + best));
}
function speechScore(target, heard) { if (!heard) return 0; const a = toJamo(target), b = toJamo(heard); return 2 * matchLen(a, b) / (a.length + b.length); }
function mouthHeights(p) {
  const [rx, ry] = p[R_EYE], [lx, ly] = p[L_EYE];
  const iod = Math.hypot(lx - rx, ly - ry), ang = Math.atan2(ly - ry, lx - rx), cx = (rx + lx) / 2, cy = (ry + ly) / 2;
  const c = Math.cos(-ang), s = Math.sin(-ang);
  const relY = ([x, y]) => ((x - cx) * s + (y - cy) * c) / iod;
  return [relY(p[R_MOUTH]), relY(p[L_MOUTH])];
}
function judgeFace(neutral, smile) {
  if (neutral.length < 5 || smile.length < 5) return { ok: null, reason: '얼굴을 찾지 못했어요' };
  const nR = median(neutral.map(m => m[0])), nL = median(neutral.map(m => m[1]));
  const lifts = smile.map(m => [nR - m[0], nL - m[1]]).sort((a, b) => (b[0] + b[1]) - (a[0] + a[1]));
  const top = lifts.slice(0, Math.max(3, Math.floor(lifts.length / 3)));
  const liftR = median(top.map(t => t[0])), liftL = median(top.map(t => t[1]));
  const big = Math.max(liftR, liftL), small = Math.min(liftR, liftL);
  if (big < CFG.FACE_MIN_LIFT) return { ok: null, reason: '웃음이 감지되지 않았어요' };
  const ratio = Math.max(small, 0) / big, weak = liftR < liftL ? '오른쪽' : '왼쪽', ok = ratio >= CFG.FACE_RATIO_MIN;
  return { ok, ratio, weak, reason: `입꼬리 좌우 비율 ${Math.round(ratio * 100)}%` + (ok ? '' : ` (${weak} 입꼬리가 덜 올라감)`) };
}

      const msg2 = `${HK[h]}에 기기를 쥐고, 엄지로 ${TAP_BTN[h]}을 최대한 빨리 누르세요.`;
      say(`${HK[h]} 검사입니다. 엄지로 ${TAP_BTN[h].slice(0, 4)}을 최대한 빨리 누르세요.`);
      await countdown(PREP_TAP, `${HK[h]} 엄지로 ${TAP_BTN[h].slice(0, 4)} 연타`, msg2, id);
      say('시작!');
      run.phase = 'TAP'; run.hand = h; run.t0 = performance.now(); send('TAP_' + h);
      for (let t = 0; t < TEST * 10; t++) {
        if (sim) run.taps[h] = Math.floor((t / 10) * ((sim === 'left' && h === 'L') ? 2.6 : (h === 'R' ? 5.6 : 5.0)));
        setTap(h, run.taps[h]);
        show('run', `${HK[h]}  ${run.taps[h]}회`, '계속 빠르게 누르세요!', `측정 ${(TEST - t / 10).toFixed(1)}s`);
        await wait(100, id);
      }
      run.phase = null; send('STOP');
      await wait(1200, id);
    }
    if (Math.max(run.taps.L, run.taps.R) === 0) { stopAll(true); setStep(0); show('warn', '버튼 입력이 없어요', '왼손 L은 B 버튼(화면 오른쪽), 오른손 R은 A 버튼(화면 왼쪽)을 눌러 주세요.'); say('버튼 입력이 없어요'); return; }
    note(`두드리기  왼 ${run.taps.L}회   오른 ${run.taps.R}회`);

    setStep(3);
    run.answers.face = await faceStep(id);
    await wait(1500, id);
    setStep(4);
    run.answers.speech = await speechStep(id);
    await wait(1500, id);
    setStep(5);
    finish();
  } catch (e) {
    if (e.message !== 'stopped') { console.error(e); stopAll(true); show('warn', '오류가 났어요', e.message); }
  }
}
function simDrift() {
  if (run.phase !== 'DRIFT') return;
  const t = (performance.now() - run.t0) / 1000, k = Math.min(t / TEST, 1);
  for (const h of ['L', 'R']) {
    let p, r;
    if (run.sim === 'left' && h === 'L') { p = -20 * k ** 1.3; r = 24 * k; } else { p = -1.5 * k; r = 1.0 * k; }
    onLine(`D,${h},${(p + gauss(0.6) + (h === 'L' ? 3 : -2)).toFixed(1)},${(r + gauss(0.6)).toFixed(1)}`);
  }
}

function preferBuiltin(devs) {

async function openMic() {

    s.getTracks().forEach(t => t.stop());
    s = await navigator.mediaDevices.getUserMedia({ audio: { deviceId: { exact: want.deviceId } } });
  }
  return s;
}

async function faceStep(id) {
  if (!$('optAuto').checked) return askManual('face', id);

  }
}
function measureFace(id) {
  return new Promise((res, rej) => {
    const v = $('video'), cv = $('camCanvas'), c = cv.getContext('2d');
    const data = { neutral: [], smile: [] };
    let phase = 'prep', t0 = performance.now(), lastTs = -1;
    show('step', '카메라를 정면으로 보세요', '가만히 계시면 평소 얼굴을 먼저 잽니다.', '얼굴 · 자동');
    say('얼굴 확인입니다. 카메라를 정면으로 보고 가만히 계세요.');
    const tick = () => {
      if (!alive(id)) { rej(new Error('stopped')); return; }
      const w = v.videoWidth || 640, h = v.videoHeight || 480;
      if (cv.width !== w) { cv.width = w; cv.height = h; }
      c.save(); c.translate(w, 0); c.scale(-1, 1); c.drawImage(v, 0, 0, w, h);
      const now = performance.now();
      if (now > lastTs) {
        lastTs = now;
        const out = landmarker.detectForVideo(v, now);
        const lm = out.faceLandmarks && out.faceLandmarks[0];
        if (lm) {
          const P = {}; for (const i of [R_EYE, L_EYE, R_MOUTH, L_MOUTH]) P[i] = [lm[i].x * w, lm[i].y * h];
          if (phase === 'neutral' || phase === 'smile') data[phase].push(mouthHeights(P));
          c.strokeStyle = 'rgba(255,255,255,.85)'; c.lineWidth = 1.5; c.setLineDash([6, 5]);
          c.beginPath(); c.moveTo(...P[R_EYE]); c.lineTo(...P[L_EYE]); c.stroke(); c.setLineDash([]);
          c.fillStyle = '#ffffff'; for (const i of [R_EYE, L_EYE]) { c.beginPath(); c.arc(P[i][0], P[i][1], 4, 0, 7); c.fill(); }
          c.fillStyle = '#e8b54a'; for (const i of [R_MOUTH, L_MOUTH]) { c.beginPath(); c.arc(P[i][0], P[i][1], 6.5, 0, 7); c.fill(); }
        }
      }
      c.restore();
      const t = (now - t0) / 1000;
      if (phase === 'prep' && t > 3.5) { phase = 'neutral'; t0 = now; show('run', '그대로 가만히', '평소 얼굴을 재고 있어요 (2초)', '얼굴 · 측정'); }
      else if (phase === 'neutral' && t > 2.5) { phase = 'smile'; t0 = now; show('run', "'이~' 하고 크게 웃으세요", '입꼬리를 양쪽으로 쭉 올려 주세요 (3초)', '얼굴 · 측정'); say('이 하고 크게 웃으세요.'); }
      else if (phase === 'smile' && t > 3.5) { camLoop = null; res(judgeFace(data.neutral, data.smile)); return; }
      camLoop = requestAnimationFrame(tick);
    };
    camLoop = requestAnimationFrame(tick);
  });
}

// ── 말 ──
let micNoted = false;
function listen(ms, track) {
  return new Promise((res, rej) => {
    const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
    const r = new SR(); r.lang = 'ko-KR'; r.interimResults = false; r.maxAlternatives = 1; r.continuous = true;
    let got = [], failed = false;
    r.onresult = e => { got = Array.from(e.results).map(x => x[0].transcript); };
    r.onerror = e => { if (e.error !== 'no-speech' && e.error !== 'aborted') { failed = true; rej(new Error(e.error)); } };
    r.onend = () => { if (!failed) res(got.join(' ').trim()); };
    let usedTrack = false;
    if (track) { try { r.start(track); usedTrack = true; } catch (e) { usedTrack = false; } }   // 최신 크롬: 고른 마이크로 인식
    if (!usedTrack) r.start();                                                                 // 지원 안 하면 크롬 기본 마이크
    if (!micNoted) {
      micNoted = true;
      note(usedTrack ? `마이크      ${track.label}` : '마이크      크롬 기본 마이크 사용 (아이폰이면 크롬 설정에서 맥북 마이크로 변경)');
    }
    setTimeout(() => { try { r.stop(); } catch (e) {} }, ms);
  });
}
async function speechStep(id) {
  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  if (!$('optAuto').checked || !SR) { if ($('optAuto').checked) note('말: 이 브라우저는 음성 인식을 지원하지 않아 직접 판단'); return askManual('speech', id); }
  for (let attempt = 0; attempt < 2; attempt++) {
    const sentence = SENTENCES[Math.floor(Math.random() * SENTENCES.length)];
    show('step', `“${sentence}.”`, '‘삐’ 소리가 나면 이 문장을 따라 읽어 주세요.', '말 · 자동');
    await say(`말 확인입니다. 삐 소리가 나면 따라 읽어 주세요. ${sentence}.`);
    if (!alive(id)) throw new Error('stopped');
    await sleep(300); beep();
    show('run', `“${sentence}.”`, '지금 읽어 주세요. 6초 동안 듣고 있어요.', '말 · 듣는 중');
    let heard = '', mic = null;
    try {
      try { mic = await openMic(); } catch (e) { mic = null; }       // 맥북 마이크를 열어서 인식에 넘김
      heard = await listen(6000, mic ? mic.getAudioTracks()[0] : null);
    }
    catch (e) { note(`말: 음성 인식 불가 (${e.message === 'not-allowed' ? '마이크 권한 거부' : e.message}) → 직접 판단`); return askManual('speech', id); }
    finally { if (mic) mic.getTracks().forEach(t => t.stop()); }
    if (!alive(id)) throw new Error('stopped');
    const score = speechScore(sentence, heard);
    if (score < CFG.SPEECH_MIN && attempt === 0) {
      const msg = heard ? '한 번 더 또박또박 읽어 주세요' : '잘 안 들렸어요';
      show('warn', msg, `들린 말: “${heard || '(없음)'}”`, '말 · 다시'); await say(`${msg}. 다른 문장으로 한 번 더 해 볼게요.`); continue;
    }
    const ok = score >= CFG.SPEECH_MIN;
    run.speechScore = score; run.heard = heard;
    note(`말        발음 정확도 ${Math.round(score * 100)}%   들린 말 “${heard || '(없음)'}”`);
    if (ok) { show('ok', `발음 정확도 ${Math.round(score * 100)}%`, `들린 말: “${heard}”`, '말 · 자동'); say('말도 정상입니다.'); }
    else { show('warn', `발음 정확도 ${Math.round(score * 100)}%`, `들린 말: “${heard || '(없음)'}”\n말이 어눌하게 들려요.`, '말 · 자동'); say('발음이 어눌하게 들려요.'); }
    return ok;
  }
}

function finish() {
  run.active = false; $('btnStart').disabled = false;

              tapL: run.taps.L, tapR: run.taps.R, faceOk: run.answers.face !== false, speechOk: run.answers.speech !== false };
  const res = judge(m, normals());
  const now = Date.now();
  if (res.level === '재검' && lastYellow && now - lastYellow < 15 * 60 * 1000) { res.level = '위험'; res.reasons.push('재검에서도 이상 신호'); }
  lastYellow = res.level === '재검' ? now : null;
  const prevNormal = lastNormalRec();
  hist().push({ ts: now, m, level: res.level, reasons: res.reasons, faceRatio: run.faceRatio, speechScore: run.speechScore, heard: run.heard, sim: run.sim });
  save(); renderResult(m, res); renderHistory();
  if (res.level === '정상') {

    say('정상입니다. 마지막 정상 시각이 기록되었어요.'); send('OK');

    say('이상 신호가 하나 있어요. 1분 쉬고 다시 체크해 주세요.'); send('WARN');
  } else {
    send('ALARM'); say('뇌졸중이 의심됩니다. 지금 바로 119, 베트남에서는 115에 전화하세요.');
    const tick = () => {
      let t;
      if (prevNormal) {
        const el = Date.now() - prevNormal.ts, rem = CFG.GOLDEN_H * 3600e3 - el;
        t = `마지막 정상 확인 ${hm(new Date(prevNormal.ts))} · ${fmtDur(el)} 전\n남은 골든타임(추정) ${rem > 0 ? hms(rem) : '지남 — 그래도 지금 바로 병원으로'}`;

      renderDial();
    };
    tick(); alarmTimer = setInterval(tick, 1000);
  }
  renderDial();
}

def angle_change(samples):
    """samples: [(시간, 처짐각, 회전각)] → (처짐 변화, 회전 변화)
    처음 1초 평균 자세와 마지막 1초 평균 자세의 차이 (순간 흔들림은 평균으로 걸러냄)"""
    if len(samples) < 6:
        return None
    start, end = samples[0][0], samples[-1][0]
    # 처음 1초는 팔을 올리며 자세를 잡는 시간 → 1~2초 평균을 기준 자세로 사용
    first = [s for s in samples if start + 1.0 <= s[0] < start + 2.0]
    if len(first) < 3:
        first = [s for s in samples if s[0] <= start + 1.0]
    last = [s for s in samples if s[0] >= end - 1.0]
    dp = statistics.mean(s[1] for s in last) - statistics.mean(s[1] for s in first)
    dr = statistics.mean(s[2] for s in last) - statistics.mean(s[2] for s in first)
    return abs(dp), abs(dr)


def derive(m):
    """측정값 → 판정에 쓰는 4가지 지표"""
    fast = max(m["tap_l"], m["tap_r"])
    return {
        "drop_max": max(m["drop_l"], m["drop_r"]),
        "diff": abs(m["drop_l"] - m["drop_r"]),
        "rot_max": max(m["rot_l"], m["rot_r"]),
        "tap_ratio": min(m["tap_l"], m["tap_r"]) / fast if fast else 0.0,
    }


def judge(m, history):
    """m: 오늘 측정값, history: 같은 사람의 과거 '정상' 기록 지표 목록
    반환: (등급 '정상'/'재검'/'위험', 이유 목록, 지표)"""
    d = derive(m)
    weak = "왼" if m["drop_l"] >= m["drop_r"] else "오른"
    rot = "왼" if m["rot_l"] >= m["rot_r"] else "오른"
    slow = "왼" if m["tap_l"] <= m["tap_r"] else "오른"
    text = {
        "drop_max": f"{weak}팔 처짐 {d['drop_max']:.0f}°",
        "diff": f"좌우 팔 높이 차이 {d['diff']:.0f}°",
        "rot_max": f"{rot}손 안쪽 회전 {d['rot_max']:.0f}°",
        "tap_ratio": f"{slow}손 두드리기 느림 ({d['tap_ratio']:.0%})",
    }
    use_base = len(history) >= BASELINE_MIN
    flags = []

    # 각도 지표: 클수록 나쁨
    for key, limit in (("drop_max", LIMIT_DROP), ("diff", LIMIT_DIFF), ("rot_max", LIMIT_ROT)):
        x = d[key]
        if use_base:
            vals = [h[key] for h in history]
            mu, sd = statistics.mean(vals), max(statistics.pstdev(vals), 2.0)
            bad = (x - mu) / sd > Z_LIMIT or x > limit * 2      # 평소보다 크게 벗어남 / 안전 상한
        else:
            bad = x > limit
        if bad:
            flags.append(key)

    # 두드리기 비율: 작을수록 나쁨
    x = d["tap_ratio"]
    if use_base:
        vals = [h["tap_ratio"] for h in history]
        mu, sd = statistics.mean(vals), max(statistics.pstdev(vals), 0.05)
        bad = (mu - x) / sd > Z_LIMIT or x < 0.5
    else:
        bad = x < LIMIT_TAP_RATIO
    if bad:
        flags.append("tap_ratio")

    reasons = [text[k] for k in flags]
    if not m["speech_ok"]:
        reasons.insert(0, "말이 어눌함")
    if not m["face_ok"]:
        reasons.insert(0, "얼굴 한쪽 처짐")

    if not m["face_ok"] or not m["speech_ok"] or len(flags) >= 2:
        level = "위험"
    elif flags:
        level = "재검"
    else:
        level = "정상"
    d["mode"] = f"개인 기준선 적용 (정상 기록 {len(history)}회)" if use_base else "기본 기준 적용"
    return level, reasons, d

class Speaker:
    """운영체제 기본 음성으로 한국어를 읽는다 (추가 설치 필요 없음)"""

    def __init__(self):
        self.proc = None
        self.mac_voice = None
        if IS_MAC:
            try:
                voices = subprocess.run(["say", "-v", "?"], capture_output=True, text=True).stdout
                if "Yuna" in voices:
                    self.mac_voice = "Yuna"
            except OSError:
                pass

    def say(self, text):
        print("[음성]", text)
        if self.proc and self.proc.poll() is None:
            self.proc.terminate()
        try:
            if IS_MAC:
                cmd = ["say"] + (["-v", self.mac_voice] if self.mac_voice else []) + [text]
                self.proc = subprocess.Popen(cmd)
            elif IS_WIN:
                safe = text.replace("'", "''")
                ps = ("Add-Type -AssemblyName System.Speech; "
                      "$s = New-Object System.Speech.Synthesis.SpeechSynthesizer; "
                      "try { $s.SelectVoiceByHints([System.Speech.Synthesis.VoiceGender]::NotSet, "
                      "[System.Speech.Synthesis.VoiceAge]::NotSet, 0, "
                      "[System.Globalization.CultureInfo]'ko-KR') } catch {}; "
                      f"$s.Speak('{safe}')")
                self.proc = subprocess.Popen(["powershell", "-NoProfile", "-Command", ps],
                                             creationflags=0x08000000)
        except OSError:
            pass

    def wait(self, timeout=15):
        """지금 읽고 있는 안내가 끝날 때까지 기다림 (녹음 전에 사용)"""
        end = time.time() + timeout
        while self.proc and self.proc.poll() is None and time.time() < end:
            time.sleep(0.1)

        self.lines = {"L": None, "R": None}

    def plot(self, h):
        s = self.samples[h]
        if len(s) < 2:
            return
        base = s[0][1]
        pts = []
        for t, p, _ in s:
            pts += [self.gx(t), self.gy(p - base)]
        if self.lines[h] is None:
            color = "#2563eb" if h == "L" else "#ea580c"
            self.lines[h] = self.canvas.create_line(*pts, fill=color, width=3)
        else:
            self.canvas.coords(self.lines[h], *pts)

        self.lbl_sub.config(text=sub)

    def countdown(self, n, title, then):
        if not self.running:
            return
        if n <= 0:
            then()
            return
        self.lbl_main.config(text=f"{title}   {n}")
        self.root.after(1000, lambda: self.countdown(n - 1, title, then))

    def begin_phase(self, phase, hand, cmd):
        self.phase, self.tap_hand, self.t0 = phase, hand, time.time()
        self.send(cmd)
        if self.sim:
            self.root.after(100, self.sim_tick)

    def end_phase(self):
        self.phase = None
        self.send("STOP")

    def drift_run(self):
        self.show("run", f"그대로 버티세요 ({TEST_SEC}초)", "눈을 감고, 팔을 움직이지 마세요")
        self.speaker.say("시작. 그대로 버티세요.")
        self.begin_phase("DRIFT", None, "DRIFT")
        self.root.after(TEST_SEC * 1000, self.drift_end)

    def drift_end(self):
        self.end_phase()
        res = {h: angle_change(self.samples[h]) for h in "LR"}
        missing = [HAND_KO[h] for h in "LR" if res[h] is None]
        if missing:
            self.abort(f"{', '.join(missing)} 기기 신호가 없어요", "손 마이크로비트 전원과 라디오 그룹(23)을 확인하세요")
            return
        self.speaker.say("끝. 팔을 내리세요.")
        self.drift = res
        self.detail.append(f"팔 처짐   왼 {res['L'][0]:.1f}°  /  오른 {res['R'][0]:.1f}°")
        self.detail.append(f"손 회전   왼 {res['L'][1]:.1f}°  /  오른 {res['R'][1]:.1f}°")
        self.lbl_detail.config(text="\n".join(self.detail))
        self.root.after(2500, lambda: self.tap_prep("R"))

    def tap_prep(self, hand):
        if not self.running:
            return
        title = f"2단계  {HAND_KO[hand]} 두드리기"
        self.show("step", title, f"{HAND_KO[hand]}에 기기를 쥐고, 엄지로 {TAP_BTN[hand]}을 최대한 빨리 누르세요 ({TEST_SEC}초)")
        self.speaker.say(f"{HAND_KO[hand]} 검사입니다. 엄지로 {TAP_BTN[hand][:4]}을 최대한 빨리 누르세요.")
        self.countdown(PREP_TAP, title, lambda: self.tap_run(hand))

    def tap_run(self, hand):
        self.show("run", f"{HAND_KO[hand]}  0회", "계속 빠르게 누르세요!")
        self.speaker.say("시작!")
        self.begin_phase("TAP", hand, f"TAP_{hand}")
        self.root.after(TEST_SEC * 1000, lambda: self.tap_end(hand))

    def tap_end(self, hand):
        self.end_phase()
        if hand == "R":
            self.root.after(1500, lambda: self.tap_prep("L"))
            return
        if max(self.taps.values()) == 0:
            self.abort("버튼 입력이 없어요", "왼손 L은 B 버튼, 오른손 R은 A 버튼을 누르는지, 전원이 켜져 있는지 확인하세요")
            return
        self.detail.append(f"두드리기  왼 {self.taps['L']}회  /  오른 {self.taps['R']}회")
        self.lbl_detail.config(text="\n".join(self.detail))
        self.root.after(1500, self.ask_face)

    def face_auto(self, retry=False, tries=0):
        if not self.running:
            return
        perm = ac.camera_permission()
        if perm == "ask":                       # 아직 안 물어봤으면 맥 허용 창을 띄우고 답을 기다림
            if tries == 0:
                ac.request_camera_permission()
                self.show("warn", "카메라 허용이 필요해요", "맥에 뜬 창에서 [허용]을 눌러 주세요")
                self.speaker.say("카메라 사용을 허용해 주세요.")
            if tries < 30:
                self.lbl_main.config(text=f"카메라 허용이 필요해요   {30 - tries}")
                self.root.after(1000, lambda: self.face_auto(retry, tries + 1))
                return
        elif perm == "denied":
            self.note("얼굴: 카메라 권한 꺼짐 (시스템 설정 → 개인정보 보호 및 보안 → 카메라) → 직접 판단")
            self.show("warn", "카메라 권한이 꺼져 있어요",
                      "시스템 설정 → 개인정보 보호 및 보안 → 카메라 → '터미널' 켜기 → 프로그램 다시 실행")
            self.speaker.say("카메라 권한이 꺼져 있어요. 이번에는 직접 판단으로 할게요.")
            self.root.after(3500, self.ask_face_manual)
            return
        try:
            if self.face is None:
                self.face = ac.FaceChecker()
            self.face.close()
            opened = self.face.open()
        except Exception as e:
            opened, self.face = False, None
            print("[카메라 오류]", e)
        if not opened:
            if tries < 15:      # 맥의 카메라 허용 창을 누를 시간을 줌 (15초)
                if tries == 0:
                    self.show("warn", "카메라 허용이 필요해요",
                              "맥에 뜬 창에서 [허용]을 누르세요.\n창이 없으면: 시스템 설정 → 개인정보 보호 및 보안 → 카메라 → '터미널' 켜기")
                    self.speaker.say("카메라 사용을 허용해 주세요.")
                self.lbl_main.config(text=f"카메라 허용이 필요해요   {15 - tries}")
                self.root.after(1000, lambda: self.face_auto(retry, tries + 1))
                return
            self.note("얼굴: 카메라 권한이 없어 직접 판단으로 전환")
            self.ask_face_manual()
            return
        self.q_key = "face_auto"
        self.face_data = {"neutral": [], "smile": []}
        self.canvas.pack_forget()
        self.cam_label.pack(anchor="w", pady=4, before=self.lbl_detail)
        self.face_phase, self.phase_t0, self.face_retry = "prep", time.time(), retry
        cam = getattr(self.face, "name", "") or "기본 카메라"
        self.show("step", "3단계  얼굴 확인 (자동)", f"카메라를 정면으로 보고 가만히 계세요\n(사용 중: {cam})")
        self.speaker.say("얼굴 확인입니다. 카메라를 정면으로 보고 가만히 계세요.")
        self.cam_tick()

    def cam_tick(self):
        if self.q_key != "face_auto" or not self.running:
            return
        img, m = self.face.read()
        if img is not None:
            pic = Image.fromarray(img)
            pic.thumbnail((G_W, 340))
            self.cam_photo = ImageTk.PhotoImage(pic)
            self.cam_label.config(image=self.cam_photo)
        t = time.time() - self.phase_t0
        ph = self.face_phase
        if m and ph in ("neutral", "smile"):
            self.face_data[ph].append(m)
        if ph == "prep" and t > 3.5:
            self.face_phase, self.phase_t0 = "neutral", time.time()
            self.show("run", "그대로 가만히", "평소 얼굴을 재는 중 (2초)")
        elif ph == "neutral" and t > 2.5:
            self.face_phase, self.phase_t0 = "smile", time.time()
            self.show("run", "'이~' 하고 크게 웃으세요!", "입꼬리를 양쪽으로 쭉 올려 주세요 (3초)")
            self.speaker.say("이 하고 크게 웃으세요.")
        elif ph == "smile" and t > 3.5:
            self.face_done()
            return
        self.root.after(40, self.cam_tick)

    def hide_cam(self):
        if self.face:
            self.face.close()
        self.cam_label.pack_forget()
        self.canvas.pack(anchor="w", pady=4, before=self.lbl_detail)

    def face_done(self):
        self.hide_cam()
        r = ac.judge_face(self.face_data["neutral"], self.face_data["smile"])
        if r["ok"] is None:
            if not self.face_retry:
                self.speaker.say(f"{r['reason']}. 한 번 더 해 볼게요.")
                self.show("warn", r["reason"], "한 번 더 해 볼게요")

            else:
                self.note(f"얼굴: {r['reason']} → 직접 판단으로 전환")
                self.ask_face_manual()
            return
        self.q_key = None
        self.face_ratio = r["ratio"]
        self.answers["face"] = r["ok"]
        self.note(f"얼굴(자동)  {r['reason']}")
        if r["ok"]:
            self.show("ok", "얼굴 대칭 정상", r["reason"])

        else:
            self.show("warn", "얼굴 한쪽이 처져 보여요", r["reason"])
            self.speaker.say(f"{r['weak']} 입꼬리가 덜 올라갔어요.")
        self.root.after(2200, self.ask_speech)

    def speech_auto(self, retry=False):
        self.q_key = "speech_auto"
        self.sentence = ac.pick_sentence()
        self.show("step", "4단계  말 확인 (자동)", f"‘삐’ 소리가 나면 따라 읽어 주세요\n“{self.sentence}.”")
        self.speaker.say(f"말 확인입니다. 삐 소리가 나면 따라 읽어 주세요. {self.sentence}.")
        threading.Thread(target=self.speech_worker, args=(retry,), daemon=True).start()

    def speech_worker(self, retry):
        """별도 스레드: 안내가 끝나면 녹음 → 음성 인식 (화면 작업은 uiq로 넘김)"""
        self.speaker.wait(15)
        time.sleep(0.3)
        self.speaker.beep()
        self.uiq.put(lambda: self.show("run", "지금 읽어 주세요", f"“{self.sentence}.”   (5초 녹음 중)"))
        heard, rms, err = "", 0.0, None
        try:
            raw, rms = ac.record()
            self.uiq.put(lambda: self.show("step", "듣고 있어요…", "발음을 분석하는 중"))
            heard = ac.recognize(raw) if rms > 120 else ""
        except Exception as e:
            err = str(e)
        self.uiq.put(lambda: self.speech_done(heard, rms, err, retry))

    def speech_done(self, heard, rms, err, retry):
        if self.q_key != "speech_auto" or not self.running:
            return
        if err:
            print("[음성 인식 오류]", err)
            self.note("말: 마이크·인터넷 문제로 직접 판단으로 전환")
            self.ask_speech_manual()
            return
        score = ac.speech_score(self.sentence, heard)
        if score < ac.SPEECH_SCORE_MIN and not retry:      # 한 번은 다시 기회
            msg = "잘 안 들렸어요" if not heard else "한 번 더 또박또박 읽어 주세요"
            self.show("warn", msg, f"들린 말: “{heard or '(없음)'}”")
            self.speaker.say(f"{msg}. 다른 문장으로 한 번 더 해 볼게요.")
            self.root.after(3500, lambda: self.speech_auto(retry=True))
            return
        ok = score >= ac.SPEECH_SCORE_MIN
        self.q_key = None
        self.speech_score, self.heard = score, heard
        self.answers["speech"] = ok
        self.note(f"말(자동)  발음 정확도 {score:.0%}  ·  들린 말: “{heard or '(없음)'}”")
        if ok:
            self.show("ok", f"발음 정확도 {score:.0%}", f"들린 말: “{heard}”")
            self.speaker.say("말도 정상입니다.")
        else:
            self.show("warn", f"발음 정확도 {score:.0%}", f"들린 말: “{heard or '(없음)'}”  → 말이 어눌하게 들려요")
            self.speaker.say("발음이 어눌하게 들려요.")
        self.root.after(2000, self.finish)

    def answer(self, ok, key=None):
        if not self.running or self.q_key is None or (key and key != self.q_key):
            return
        done = self.q_key
        self.answers[done] = ok
        self.q_key = None
        self.btn_yes.config(state="disabled")
        self.btn_no.config(state="disabled")
        if done == "face":
            self.ask_speech()
        else:
            self.finish()

    def finish(self):
        self.running = False
        self.btn_start.config(state="normal")
        m = {"drop_l": self.drift["L"][0], "drop_r": self.drift["R"][0],
             "rot_l": self.drift["L"][1], "rot_r": self.drift["R"][1],
             "tap_l": self.taps["L"], "tap_r": self.taps["R"],
             "face_ok": self.answers.get("face", True), "speech_ok": self.answers.get("speech", True)}
        level, reasons, d = judge(m, self.baseline_history())

        now = time.time()
        if level == "재검" and self.last_yellow and now - self.last_yellow < 15 * 60:
            level = "위험"
            reasons.append("재검에서도 이상 신호")
        self.last_yellow = now if level == "재검" else None

        self.db.execute(
            "INSERT INTO checks (ts, user, drop_l, drop_r, rot_l, rot_r, tap_l, tap_r, face_ok, speech_ok, "
            "result, reasons, face_ratio, speech_score, heard) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
            (datetime.now().strftime("%Y-%m-%d %H:%M:%S"), self.uname(), m["drop_l"], m["drop_r"],
             m["rot_l"], m["rot_r"], m["tap_l"], m["tap_r"], int(m["face_ok"]), int(m["speech_ok"]),
             level, " / ".join(reasons), self.face_ratio, self.speech_score, self.heard))
        self.db.commit()
        self.detail.append(f"판정 기준: {d['mode']}")
        self.lbl_detail.config(text="\n".join(self.detail))
        self.refresh()

        if level == "정상":
            self.show("ok", "정상입니다 ✔",
                      f"마지막 정상 확인 시각이 {datetime.now():%H:%M}로 기록되었어요. 오늘도 건강하세요!")
            self.speaker.say("정상입니다. 마지막 정상 시각이 기록되었어요.")
            self.send("OK")
        elif level == "재검":
            self.show("warn", "재검이 필요해요",
                      f"{' · '.join(reasons)}\n1분 쉬고 다시 체크하세요. 다시 이상이 나오면 바로 병원에 연락하세요.")
            self.speaker.say("이상 신호가 하나 있어요. 1분 쉬고 다시 체크해 주세요.")
            self.send("WARN")
        else:

        m = None
        if res.multi_face_landmarks:
            lm = res.multi_face_landmarks[0].landmark
            pts = {i: (lm[i].x * w, lm[i].y * h) for i in (R_EYE, L_EYE, R_MOUTH, L_MOUTH)}
            m = mouth_heights(pts)
            for i in (R_EYE, L_EYE):
                cv2.circle(rgb, tuple(int(v) for v in pts[i]), 4, (80, 160, 255), -1)
            for i in (R_MOUTH, L_MOUTH):
                cv2.circle(rgb, tuple(int(v) for v in pts[i]), 6, (255, 90, 40), -1)
            cv2.line(rgb, tuple(int(v) for v in pts[R_EYE]), tuple(int(v) for v in pts[L_EYE]), (80, 160, 255), 2)
        return cv2.flip(rgb, 1), m       # 거울처럼 보이게 좌우 반전 (측정은 반전 전에 끝남)


def mouth_heights(pts):
    """눈꼬리 선을 수평으로 돌린 뒤, 눈 선에서 각 입꼬리까지의 세로 거리 / 눈 사이 거리"""
    (rx, ry), (lx, ly) = pts[R_EYE], pts[L_EYE]
    iod = math.hypot(lx - rx, ly - ry)
    ang = math.atan2(ly - ry, lx - rx)
    cx, cy = (rx + lx) / 2, (ry + ly) / 2
    cos, sin = math.cos(-ang), math.sin(-ang)

    def rel_y(p):
        x, y = p[0] - cx, p[1] - cy
        return (x * sin + y * cos) / iod      # 회전 후 y (아래가 +)

    return rel_y(pts[R_MOUTH]), rel_y(pts[L_MOUTH])


def judge_face(neutral, smile):
    """neutral, smile: [(오른쪽 높이, 왼쪽 높이), ...]
    반환: dict(ok, ratio, lift_r, lift_l, weak, reason)"""
    if len(neutral) < 5 or len(smile) < 5:
        return {"ok": None, "reason": "얼굴을 찾지 못했어요"}
    n_r = float(np.median([m[0] for m in neutral]))
    n_l = float(np.median([m[1] for m in neutral]))
    lifts = sorted(((n_r - m[0], n_l - m[1]) for m in smile), key=lambda t: t[0] + t[1], reverse=True)
    top = lifts[:max(3, len(lifts) // 3)]           # 가장 활짝 웃은 순간들
    lift_r = float(np.median([t[0] for t in top]))
    lift_l = float(np.median([t[1] for t in top]))
    big, small = max(lift_r, lift_l), min(lift_r, lift_l)
    if big < FACE_MIN_LIFT:
        return {"ok": None, "reason": "웃음이 감지되지 않았어요", "lift_r": lift_r, "lift_l": lift_l}
    ratio = max(small, 0.0) / big
    weak = "오른쪽" if lift_r < lift_l else "왼쪽"
    ok = ratio >= FACE_RATIO_MIN
    reason = f"입꼬리 좌우 비율 {ratio:.0%}" + ("" if ok else f" ({weak} 입꼬리가 덜 올라감)")
    return {"ok": ok, "ratio": ratio, "lift_r": lift_r, "lift_l": lift_l, "weak": weak, "reason": reason}

# 더블클릭하면 골든타임 지킴이 프로그램이 켜집니다 (최신 파이썬 3.12 + 카메라·음성 인식 라이브러리)
cd "$(dirname "$0")"
"$HOME/.local/bin/uv" run --python 3.12 \
  --with pyserial --with "mediapipe==0.10.21" --with opencv-python --with pillow \
  --with sounddevice --with soundfile --with SpeechRecognition --with pyobjc-framework-AVFoundation \
  python golden_time.py "$@"
